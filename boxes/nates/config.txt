# Nate's box - Pi 4 + Raspiaudio Audio+ DAC HAT + 1024x600 HDMI touch
# panel (USB HID touch). Firmware config fragment; merged into the boot
# partitions' config.txt by boompi-apply-box-config.

# Raspiaudio Audio+ DAC HAT (PCM51xx-family I2S - vendor-documented
# overlay). Onboard audio off so the DAC is the only ALSA card.
dtoverlay=hifiberry-dac
dtparam=audio=off

# INA260 battery monitor on the standard I2C bus.
dtparam=i2c_arm=on

# USB Bluetooth dongle (UB500 Plus - same RTL8761BU/2357:0604 as the
# UB500 despite the name) is the only radio: onboard BT off, same
# reasoning as georges (antenna sharing stutters, and two controllers
# make boompid's adapter selection a boot-order coin flip).
dtoverlay=disable-bt
