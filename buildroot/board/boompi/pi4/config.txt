# Boompi Pi 4 boot configuration (board-generic).
# Box-specific hardware lives in the box profile - see
# docs/PROVISIONING.md and boxes/ for the bench boxes.

start_file=start4.elf
fixup_file=fixup4.dat
kernel=Image
arm_64bit=1

# TF-A BL31 as the armstub: provides PSCI so secondary CPUs can be
# parked - required by kexec (the A/B update trial boot; spin-table
# CPUs make kexec_load fail with EBUSY). TF-A also injects the psci
# nodes into the DT before starting the kernel.
armstub=bl31.bin
enable_gic=1

disable_overscan=1
disable_splash=1
boot_delay=0

# KMS graphics. The panel's mode is set via cmdline.txt (video=...):
# full KMS ignores the legacy hdmi_* config.txt settings v1 used, and
# the panel may not provide EDID.
dtoverlay=vc4-kms-v3d
max_framebuffers=2

# Onboard Bluetooth (BCM43455, UART-attached) stays enabled. Needs
# BCM4345C0.hcd from BR2_PACKAGE_BRCMFMAC_SDIO_FIRMWARE_RPI_BT; the
# kernel attaches hci0 via serdev automatically.

# Generic posture: HDMI + onboard audio. A provisioned box overrides
# from its profile (e.g. Nate's: hifiberry-dac for the Raspiaudio
# HAT, i2c_arm=on for the INA260, video= mode via cmdline fragment).

# Box-specific firmware config (display overlay, rotation, wiring,
# amp GPIO) belongs in /data/box/config.txt, not here: it is merged
# into a fenced section below by boompi-apply-box-config whenever a
# boot partition is written, and survives OS updates.
