# Boompi Pi 3 boot configuration (board-generic).
# Box-specific hardware lives in the box profile - see
# docs/PROVISIONING.md and boxes/ for the bench boxes.

kernel=Image
arm_64bit=1

# Stock boot chain: no armstub (TF-A left with the retired kexec
# mechanism), firmware-default kernel/DTB placement, spin-table SMP.
disable_overscan=1
disable_splash=1
boot_delay=0

# KMS graphics
dtoverlay=vc4-kms-v3d
max_framebuffers=2

# Generic posture: HDMI console via vc4-kms-v3d, onboard Bluetooth
# and audio left enabled. A provisioned box overrides all of this
# from its profile (e.g. George's: HyperPixel DPI panel, disable-bt
# for the USB dongle, audio=off for USB out).

# Box-specific firmware config (display overlay, rotation, wiring,
# amp GPIO) belongs in /data/box/config.txt, not here: it is merged
# into a fenced section below by boompi-apply-box-config whenever a
# boot partition is written, and survives OS updates.
