# Boompi Pi 3 box boot configuration.
# Hardware facts recovered from the v1 SD card + Phase 0 spikes
# (docs/PHASE0-PI3.md).

kernel=Image
arm_64bit=1

# TF-A BL31 armstub (armstub8.bin, auto-loaded by name for arm_64bit):
# provides PSCI so secondary CPUs can be parked - required by kexec
# (the A/B update trial boot). The rpi3 TF-A port uses fixed load
# addresses: kernel and DTB locations must match the PRELOADED_BL33_BASE
# and RPI3_PRELOADED_DTB_BASE build options in the board fragment.
#
# The DTB must sit ABOVE the kernel image: with a 20MB Image at 32MB,
# a DTB at the TF-A-documented 0x01000000 hangs start.elf before it
# even shows the splash (bench-bisected on firmware 2026-05-21; the
# firmware appears to stage the kernel low in RAM before moving it to
# kernel_address, and the DTB write tramples the staging area).
kernel_address=0x02000000
device_tree_address=0x03800000

disable_overscan=1
disable_splash=1
boot_delay=0

# KMS graphics
dtoverlay=vc4-kms-v3d
max_framebuffers=2

# HyperPixel 4.0 panel + GT911 touch, rotated (matches v1; Slint adds
# SLINT_KMS_ROTATION=270 on top - see boompi-ui.service)
dtoverlay=vc4-kms-dpi-hyperpixel4
dtparam=rotate=270,touchscreen-swapped-x-y,touchscreen-inverted-x

# USB Bluetooth dongle only; onboard radio disabled (stutters, shares
# antenna/UART). Audio out is the USB adapter, not onboard analog.
dtoverlay=disable-bt
dtparam=audio=off

# amp GPIO) belongs in /data/box/config.txt, not here: it is merged
# into a fenced section below by boompi-apply-box-config whenever a
# boot partition is written, and survives OS updates.
